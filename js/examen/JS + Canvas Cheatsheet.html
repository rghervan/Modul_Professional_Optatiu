Condicionals — switch

    console.log("Dilluns")

  case "dm":

  case "dj":

    console.log("Cap de setmana")
}

Bucles
// for clàssic
for (let i = 0; i < 5; i++) {
  console.log(i)    // 0,1,2,3,4
}

// for...of  (valors d'array)

  console.log(x)
}

// for...in  (claus d'objecte)
for (const k in obj) {
  console.log(k, obj[k])
}

// while

while (i < 3) { i++ }

// do...while  (mínim 1 iteració)

  i++
} while (i < 5)

// break → surt del bucle
// continue → salta iteració

Nombres aleatoris — Math.random
// Decimal entre 0 (inclòs) i 1 (exclòs)
Math.random()           // → 0.7341...

// Enter entre 0 i N-1
Math.floor(Math.random() * 10)  // 0..9

// Enter entre min i max (inclosos)
function randInt(min, max) {
  return Math.floor(
    Math.random() * (max - min + 1)
  ) + min
}
randInt(1, 6)   // dau: 1..6

// Decimal amb decimals fixes
(Math.random() * 100).toFixed(2)

// Element aleatori d'un array
const arr = ["a","b","c"]
arr[Math.floor(Math.random() * arr.length)]

// Color hexadecimal aleatori
"#" + Math.floor(Math.random()*0xFFFFFF)
        .toString(16).padStart(6,"0")
Canvas — configuració bàsica
<!-- HTML -->
<canvas id="c" width="400" height="300"></canvas>

// Obtenir context 2D
const c   = document.getElementById("c")
const ctx = c.getContext("2d")

// Netejar canvas
ctx.clearRect(0, 0, c.width, c.height)

// Colors i estil
ctx.fillStyle   = "#7F77DD"  // farciment
ctx.strokeStyle = "#fff"    // contorn
ctx.lineWidth   = 2
ctx.globalAlpha = 0.8       // opacitat
Canvas — formes i paths
// Rectangle
ctx.fillRect(10, 10, 100, 50)   // x,y,w,h
ctx.strokeRect(10, 10, 100, 50)

// Camí (path)
ctx.beginPath()
ctx.moveTo(50, 50)
ctx.lineTo(150, 50)
ctx.lineTo(100, 120)
ctx.closePath()
ctx.fill()   // o ctx.stroke()

// Cercle / arc
ctx.beginPath()
// arc(x, y, radi, inici, fi)
ctx.arc(100, 100, 40, 0, Math.PI * 2)
ctx.fill()

// Corba de Bézier

Canvas — text, imatge i animació
// Text
ctx.font         = "bold 16px monospace"
ctx.textAlign    = "center"   // left|right
ctx.textBaseline = "middle"
ctx.fillText("Hola!", 200, 150)
ctx.strokeText("Hola!", 200, 150)

// Animació amb requestAnimationFrame
let x = 0
function loop() {
  ctx.clearRect(0, 0, c.width, c.height)
  ctx.fillRect(x++, 100, 30, 30)
  requestAnimationFrame(loop)
}
